REGRAS DO FIRESTORE — FASE 19

COMO PUBLICAR
1. Firebase Console > Firestore Database > Regras.
2. Substitua TODAS as regras atuais pelo bloco abaixo.
3. Clique em Publicar.
4. Não altere as variáveis do Firebase Admin nem a URL do webhook Hotmart.

Estas regras não interferem no webhook: a Vercel usa Firebase Admin, que é uma
rota de servidor separada. Elas liberam somente os dados acadêmicos necessários
ao site: progresso, atividades, projetos, aulas e matrícula direta confirmada.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function owns(userId) { return signedIn() && request.auth.uid == userId; }
    function isStaff() {
      return signedIn()
        && exists(/databases/$(database)/documents/staff/$(request.auth.uid))
        && get(/databases/$(database)/documents/staff/$(request.auth.uid)).data.active == true;
    }
    function isPaidStudent() {
      return signedIn()
        && exists(/databases/$(database)/documents/students/$(request.auth.uid))
        && get(/databases/$(database)/documents/students/$(request.auth.uid)).data.enrollmentStatus == 'paid';
    }

    match /staff/{staffId} {
      allow read: if owns(staffId) || isStaff();
      allow write: if false;
    }

    match /students/{studentId} {
      allow read: if owns(studentId) || isStaff();
      allow create, update, delete: if isStaff();
    }

    match /students/{studentId}/progress/{progressId} {
      allow read: if owns(studentId) || isStaff();
      allow create, update: if owns(studentId) || isStaff();
      allow delete: if isStaff();
    }

    match /students/{studentId}/activities/{activityId} {
      allow read: if owns(studentId) || isStaff();
      allow create: if owns(studentId)
        && request.resource.data.studentId == request.auth.uid
        && request.resource.data.answer is string
        && request.resource.data.answer.size() >= 12;
      allow update: if isStaff() || (owns(studentId)
        && resource.data.status != 'graded'
        && request.resource.data.status == 'submitted'
        && request.resource.data.studentId == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['answer', 'status', 'updatedAt', 'submittedAt']));
      allow delete: if isStaff();
    }

    match /students/{studentId}/projects/{projectId} {
      allow read: if owns(studentId) || isStaff();
      allow create: if owns(studentId)
        && request.resource.data.studentId == request.auth.uid
        && request.resource.data.notes is string
        && request.resource.data.notes.size() >= 12;
      allow update: if isStaff() || (owns(studentId)
        && request.resource.data.studentId == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['driveUrl', 'notes', 'status', 'submittedAt', 'updatedAt']));
      allow delete: if isStaff();
    }

    match /courseLessons/{lessonId} {
      allow read: if isStaff() || isPaidStudent();
      allow create, update, delete: if isStaff();
    }

    match /assessments/{assessmentId} {
      allow read: if signedIn();
      allow create, update, delete: if isStaff();
      match /submissions/{studentId} {
        allow read: if owns(studentId) || isStaff();
        allow create: if owns(studentId)
          && request.resource.data.studentId == request.auth.uid
          && request.resource.data.status == 'submitted';
        allow update: if isStaff() || (owns(studentId)
          && resource.data.status == 'submitted'
          && request.resource.data.status == 'submitted'
          && request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(['answer', 'status', 'submittedAt']));
        allow delete: if isStaff();
      }
    }

    match /assessmentKeys/{assessmentId} { allow read, write: if isStaff(); }

    match /liveSessions/{sessionId} {
      allow read: if signedIn();
      allow create, update, delete: if isStaff();
      match /attendance/{studentId} {
        allow read: if owns(studentId) || isStaff();
        allow create, update, delete: if isStaff();
      }
    }

    match /mentorshipRequests/{requestId} {
      allow read: if isStaff() || (signedIn() && resource.data.studentId == request.auth.uid);
      allow create: if signedIn() && request.resource.data.studentId == request.auth.uid;
      allow update, delete: if isStaff();
    }

    // Criada somente pela equipe após confirmar boleto/PIX parcelado.
    // O servidor promove esta matrícula ao documento do aluno no primeiro login.
    match /manualEnrollments/{enrollmentId} { allow read, write: if isStaff(); }

    // Vendas e matrículas automáticas são exclusivas do Firebase Admin na Vercel.
    match /sales/{saleId} { allow read: if isStaff(); allow write: if false; }
    match /enrollments/{enrollmentId} { allow read, write: if false; }
  }
}

MÉDIA GERAL PONDERADA
- Avaliações de módulo corrigidas: 55%
- Atividades obrigatórias ao fim das aulas: 10%
- Projeto contínuo iniciado na Aula 2 do M01: 15%
- Produção avaliativa independente liberada no M19: 20%

A média fica N/A até a primeira avaliação ou trabalho receber nota do professor.
O painel calcula uma média parcial com os componentes já avaliados; os pesos são
normalizados para não penalizar o aluno por uma etapa que ainda não foi corrigida.

MATRÍCULA DIRETA
No painel do professor, abra “Matrícula direta”, informe o mesmo e-mail Google do
aluno, o plano e a data. Marque a confirmação apenas após receber o pagamento.
No primeiro login do aluno, o servidor segura essa matrícula e cria o acesso.
